% Same convention as noteadditionalspacelinestextthreshold: a LOWER value
% is what makes blnabc less reactive, not a higher one (this is backwards
% from the top-side thresholds). This is how you get back the old
% (pre-#1766) fixed-distance behavior: -2 means no additional space at all,
% matching noteadditionalspacelinestextthreshold's own documented minimum.
{
\verb|\grechangecount{additionalbottomspacenabcthreshold}{-2} (opts out)|
\grechangecount{additionalbottomspacenabcthreshold}{-2}

\begin{center}
\begin{tabular}{cccccccc}
  \lines{\nabcsnippet{2}{a_0'0[t](a'0_0||ta)}} &

  \lines{\nabcsnippet{2}{a'0[t](a'0||ta)}} &

  \lines{\nabcsnippet{2}{d[t](d||ta)}} &
  \lines{\nabcsnippet{2}{e[t](e||ta)}}
\end{tabular}
\end{center}
}

% Real-world sanity check, mirroring the above-lines-nabc-threshold.tex
% "Adorate Dominum" example: same antiphon, transposed down by 5 pitch
% steps so its lowest note lands on 'a' (requiring ledger lines below the
% staff), with the "o" syllable's pes forced upward -- toward the low note
% -- via an explicit hp pitch code (viha's mirror image: there, ha pulled a
% glyph down toward a high note above the staff; here, hp pushes a glyph up
% toward a low note below the staff). Since the collision-avoidance fix for
% explicit hX pitch codes only covers above-lines nabc so far, this is
% expected to still show a collision until/unless that fix is mirrored for
% below-lines nabc (part_blnabc).
