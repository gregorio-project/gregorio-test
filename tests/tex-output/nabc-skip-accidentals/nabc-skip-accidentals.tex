% Test for \gresetnabcskipaccidentals (single voice):
%   false (default), true (global), [1]{true}, [1]{false} (explicit reset)

\documentclass[11pt]{article}
\usepackage{fontspec}
\usepackage{geometry}
\geometry{a4paper}
\usepackage[allowdeprecated=false]{gregoriotex}
\usepackage{fullpage}
\setmainfont[
    Path = ../../../fonts/ ,
    Extension = .otf ,
    UprightFont = *-Regular ,
    UprightFeatures = { SmallCapsFont = *SC-Regular } ,
    BoldFont = *-Bold ,
    BoldFeatures = { SmallCapsFont = *SC-Bold } ,
    ItalicFont = *-Italic ,
    ItalicFeatures = { SmallCapsFont = *SC-Italic } ,
    BoldItalicFont = *-BoldItalic ,
    BoldItalicFeatures = { SmallCapsFont = *SC-BoldItalic } ,
    Ligatures = TeX
]{Alegreya}

\grechangedim{baselineskip}{55pt}{scalable}

\listfiles
\begin{document}

% 1) Default: false — neume aligns with accidental when present
\section*{\texttt{\char`\\gresetnabcskipaccidentals\{false\}} (default)}
\gresetnabcskipaccidentals{false}
\gregorioscore[a]{nabc-skip-accidentals}

% 2) Global true — neume skips leading accidentals, anchors to first square note
\section*{\texttt{\char`\\gresetnabcskipaccidentals\{true\}}}
\gresetnabcskipaccidentals{true}
\gregorioscore[a]{nabc-skip-accidentals}

% 3) Per-voice [1]{true} — explicit per-voice form, equivalent to global true
%    for a single-voice score
\section*{\texttt{\char`\\gresetnabcskipaccidentals[1]\{true\}}}
\gresetnabcskipaccidentals{false}
\gresetnabcskipaccidentals[1]{true}
\gregorioscore[a]{nabc-skip-accidentals}

% 4) Per-voice [1]{false} — explicit reset per voice
\section*{\texttt{\char`\\gresetnabcskipaccidentals[1]\{false\}}}
\gresetnabcskipaccidentals[1]{false}
\gregorioscore[a]{nabc-skip-accidentals}

\end{document}
