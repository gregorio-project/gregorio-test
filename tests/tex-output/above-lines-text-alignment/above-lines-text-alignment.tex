\newcommand{\lines}[1]{
  \begin{minipage}[b]{3cm}
    \grechangedim{parskip}{0pt}{fixed}%
    \setlength{\baselineskip}{0pt}%
    \setlength{\lineskip}{0pt}%

% Wraps text with a thin vrule dropped at its exact horizontal midpoint,
% without changing the text's own measured width (the vrule is \rlap'd, so
% it contributes zero width): useful for visually checking center alignment,
% since for <alt:c>/[alt:...;c] the wrapped text's midpoint is where
% \gre@typesettextabovelines lands, by the same offset math that puts left-
% /right-aligned text flush with the anchor's left/right edge.
\newbox\gremarkbox
\newcommand{\centermarked}[1]{%
  \setbox\gremarkbox=\hbox{#1}%
  \kern0.5\wd\gremarkbox
  \rlap{\kern-0.2pt\vrule width 0.4pt height 12pt depth 8pt}%
  \kern-0.5\wd\gremarkbox
  \box\gremarkbox
}

% Like \centermarked, but marks whichever edge/midpoint the *current*
% \gresetabovelinestextalignment default actually is (read from
% \gre@count@abovelinestextalignment: 1=left, 2=center, 3=right), so the
% mark always lands on the anchor regardless of which default is active
% when this is typeset --- unlike a fixed \centermarked, this stays
% unambiguous even without prose explaining what to expect.
\makeatletter
\newcommand{\defaultmarked}[1]{%
  \ifcase\gre@count@abovelinestextalignment\relax
  \or % 1 = left
    \vrule width 0.4pt height 15pt depth 10pt #1%
  \or % 2 = center
    \setbox\gremarkbox=\hbox{#1}%
    \kern0.5\wd\gremarkbox
    \rlap{\kern-0.2pt\vrule width 0.4pt height 15pt depth 10pt}%
    \kern-0.5\wd\gremarkbox
    \box\gremarkbox
  \or % 3 = right
    #1\vrule width 0.4pt height 15pt depth 10pt%
  \fi
}
\makeatother
